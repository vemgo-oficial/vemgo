CREATE OR REPLACE FUNCTION public.review_professional(p_professional_id uuid, p_decision text, p_rejection_reason text DEFAULT NULL::text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'public'
AS $function$
declare
  current_user_id uuid;
  professional_identity_status text;
begin
  current_user_id := auth.uid();

  if current_user_id is null then
    raise exception 'Usuário não autenticado.';
  end if;

  if not exists (
    select 1
    from public.profiles admin_profile
    where admin_profile.id = current_user_id
      and admin_profile.user_type = 'admin'
      and admin_profile.status = 'active'
  ) then
    raise exception 'Acesso negado.';
  end if;

  if p_decision not in ('approved', 'rejected') then
    raise exception 'Decisão inválida.';
  end if;

  select pr.identity_status
  into professional_identity_status
  from public.profiles pr
  where pr.id = p_professional_id;

  if professional_identity_status is null then
    raise exception 'Profissional não encontrado.';
  end if;

  if p_decision = 'approved'
     and professional_identity_status <> 'verified' then
    raise exception 'O profissional só pode ser aprovado após a identidade ser verificada.';
  end if;

  if p_decision = 'approved' then

    update public.professionals
    set
      verification_status = 'approved',
      service_status = 'active',
      approved_at = now(),
      approved_by = current_user_id,
      rejection_reason = null,
      updated_at = now()
    where id = p_professional_id;

    update public.profiles
    set
      status = 'active',
      updated_at = now()
    where id = p_professional_id;

  else

    update public.professionals
    set
      verification_status = 'rejected',
      service_status = 'inactive',
      approved_at = null,
      approved_by = null,
      rejection_reason = nullif(trim(p_rejection_reason), ''),
      updated_at = now()
    where id = p_professional_id;

  end if;

  insert into public.audit_logs (
    actor_profile_id,
    action,
    entity_type,
    entity_id,
    details
  )
  values (
    current_user_id,
    case
      when p_decision = 'approved'
        then 'professional_approved'
      else
        'professional_rejected'
    end,
    'professional',
    p_professional_id,
    jsonb_build_object(
      'decision', p_decision,
      'rejection_reason',
      nullif(trim(p_rejection_reason), '')
    )
  );
end;
$function$;
